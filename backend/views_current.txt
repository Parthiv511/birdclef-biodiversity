
from typing import Any, cast

from django.http import JsonResponse
from django.core.files.uploadedfile import UploadedFile

from rest_framework.decorators import api_view, parser_classes
from rest_framework.parsers import MultiPartParser, FormParser
from rest_framework.response import Response
from rest_framework import status

from .serializers import AudioUploadSerializer
from .ml.audio_screening import screen_uploaded_file


def health_check(request):
    """Verify that the BirdCLEF+ backend is running."""

    return JsonResponse({
        "success": True,
        "message": "BirdCLEF+ backend is running!",
        "project": "BirdCLEF+ 2026 Biodiversity Analysis",
        "status": "healthy",
    })


@api_view(["POST"])
@parser_classes([MultiPartParser, FormParser])
def upload_audio(request):
    """
    Validate an uploaded audio file and screen it for
    possible wildlife relevance.
    """

    # STEP 1: Validate the uploaded audio.
    serializer = AudioUploadSerializer(data=request.data)

    if not serializer.is_valid():
        return Response(
            {
                "success": False,
                "message": "Audio validation failed.",
                "errors": serializer.errors,
            },
            status=status.HTTP_400_BAD_REQUEST,
        )

    validated_data = cast(
        dict[str, Any],
        serializer.validated_data,
    )

    audio_file = validated_data.get("audio")

    if not isinstance(audio_file, UploadedFile):
        return Response(
            {
                "success": False,
                "message": "A valid uploaded audio file is required.",
            },
            status=status.HTTP_400_BAD_REQUEST,
        )

    # STEP 2: Run the wildlife relevance screening model.
    try:
        screening_result = screen_uploaded_file(audio_file)

    except Exception:
        # Do not expose internal exception details to API clients.
        return Response(
            {
                "success": False,
                "message": (
                    "Audio validation succeeded, but wildlife "
                    "screening failed. Please try again."
                ),
                "filename": audio_file.name,
                "size_bytes": audio_file.size,
                "analysis_status": "screening_failed",
            },
            status=status.HTTP_500_INTERNAL_SERVER_ERROR,
        )

    # STEP 3: Warn if the screening model does not identify
    # wildlife-related labels among its top predictions.
    if not screening_result["is_wildlife_related"]:
        return Response(
            {
                "success": True,
                "status": "warning",
                "is_wildlife_related": False,
                "message": screening_result["message"],
                "filename": audio_file.name,
                "size_bytes": audio_file.size,
                "analysis_status": "warning",
                "screening_predictions": (
                    screening_result["screening_predictions"]
                ),
                "note": (
                    "This is a preliminary screening result. "
                    "It does not prove that wildlife is absent."
                ),
            },
            status=status.HTTP_200_OK,
        )

    # STEP 4: Screening passed. CNN14 inference will be connected
    # in the next implementation step.
    return Response(
        {
            "success": True,
            "status": "screening_passed",
            "is_wildlife_related": True,
            "message": screening_result["message"],
            "filename": audio_file.name,
            "size_bytes": audio_file.size,
            "analysis_status": "awaiting_cnn14",
            "screening_predictions": (
                screening_result["screening_predictions"]
            ),
            "note": (
                "Wildlife screening passed. CNN14 species inference "
                "has not been connected yet."
            ),
        },
        status=status.HTTP_200_OK,
    )
